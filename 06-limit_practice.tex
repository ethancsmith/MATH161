\begin{example}

\begin{enumerate}
\item $\DS\lim_{x\to 1}\frac{|x-1|}{x-1}$
\vfill
\item $\DS\lim_{t\to 6}8(t-5)(t-7)$
\vfill
\item $\DS\lim_{y\to -3}(5-y)^{4/3}$
\vfill

\vfill
\end{enumerate}
\end{example}
