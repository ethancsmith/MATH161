\begin{remark}
Complete all derivative calculations using the definition of the derivative:
\begin{equation*}
\frac{\dee y}{\dee x} = f'(x) = \lim_{h\to 0}\frac{f(x+h)-f(x)}{h}.
\end{equation*}
\end{remark}

\begin{example}
Compute $\frac{\dee f}{\dee x}$ and its domain if $f(x) = 2x-1$.
\end{example}

\begin{example}
Determine whether the function
\begin{equation*}
f(x) = \begin{cases}
2x+\tan x & \text{if } x\ge 0,\\
x^{2} &\text{if } x<0
\end{cases}
\end{equation*}
is differentiable at $x=0$.
If it is, what is the value of the derivative?
\end{example}
